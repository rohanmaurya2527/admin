import requests,csv
from bs4 import BeautifulSoup

url="https://www.1800flowers.com/articles/flower-facts/flower-types"
s=BeautifulSoup(requests.get(url).text,"html.parser")

with open("flowers.csv","w",newline="",encoding="utf-8") as f:
    w=csv.writer(f)
    w.writerow(["Flower","Details"])
    for h in s.find_all("h2"):
        if h.get_text(strip=True)[0].isdigit():
            p=h.find_next("p")
            w.writerow([h.get_text(" ",strip=True),p.get_text(" ",strip=True) if p else ""])

print("Saved!")

import requests
from bs4 import BeautifulSoup
import csv

url = "https://books.toscrape.com/catalogue/category/books/childrens_11/index.html"

response = requests.get(url)
response.raise_for_status()

soup = BeautifulSoup(response.text, "html.parser")

books = soup.select("article.product_pod")

rating_map = {
    "One": 1,
    "Two": 2,
    "Three": 3,
    "Four": 4,
    "Five": 5
}

with open("books.csv", "w", newline="", encoding="utf-8") as file:
    writer = csv.writer(file)

    writer.writerow([
        "Title",
        "Price",
        "Availability",
        "Rating"
    ])

    for book in books:
        title = book.h3.a["title"]
        price = book.select_one(".price_color").text
        availability = book.select_one(".availability").text.strip()

        rating_class = book.select_one("p.star-rating")["class"][1]
        rating = rating_map[rating_class]

        writer.writerow([
            title,
            price,
            availability,
            rating
        ])

print("Data saved to books.csv")
